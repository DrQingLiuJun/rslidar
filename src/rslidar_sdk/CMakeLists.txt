cmake_minimum_required(VERSION 3.8)
project(rslidar_sdk VERSION 1.5.16)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rslidar_msg REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)

# Keep the deployed receiver options and XYZI point layout.
option(ENABLE_TRANSFORM "Enable coordinate transformation" OFF)
option(ENABLE_EPOLL_RECEIVE "Receive with epoll" OFF)
option(ENABLE_MODIFY_RECVBUF "Adjust receive buffer size" ON)
option(ENABLE_WAIT_IF_QUEUE_EMPTY "Wait when receive queue is empty" OFF)
option(ENABLE_DIFOP_PARSE "Parse DIFOP packets" ON)
option(ENABLE_STAMP_WITH_LOCAL "Use local timestamps" OFF)
option(ENABLE_SOURCE_PACKET_LEGACY "Legacy packet message support" OFF)
option(ENABLE_IMU_DATA_PARSE "Parse IMU data" OFF)

add_compile_definitions(ROS2_FOUND POINT_TYPE_XYZI PROJECT_PATH="${PROJECT_SOURCE_DIR}")
foreach(feature ENABLE_TRANSFORM ENABLE_EPOLL_RECEIVE ENABLE_MODIFY_RECVBUF
    ENABLE_WAIT_IF_QUEUE_EMPTY ENABLE_DIFOP_PARSE ENABLE_STAMP_WITH_LOCAL
    ENABLE_SOURCE_PACKET_LEGACY ENABLE_IMU_DATA_PARSE)
  if(${feature})
    add_compile_definitions(${feature})
  endif()
endforeach()

add_subdirectory(src/rs_driver)
find_package(rs_driver REQUIRED)
add_executable(rslidar_sdk_node node/rslidar_sdk_node.cpp src/manager/node_manager.cpp)
target_include_directories(rslidar_sdk_node PRIVATE src ${rs_driver_INCLUDE_DIRS})
target_compile_options(rslidar_sdk_node PRIVATE -Wall)
target_link_libraries(rslidar_sdk_node ${YAML_CPP_LIBRARIES} ${rs_driver_LIBRARIES})
ament_target_dependencies(rslidar_sdk_node ament_index_cpp rclcpp rslidar_msg sensor_msgs std_msgs)
install(TARGETS rslidar_sdk_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
